**Bending light around a black hole.** A point lens maps every pixel at angle θ to the source at θ − θE²/θ. That one line is the Einstein ring.

```glsl
// gravitational lens: a point lens maps each pixel at angle θ to the source at θ − θE²/θ
vec2 uv = vUv; float hole = 1.0, ring = 0.0;
if (uBH.z > 0.5) {
  vec2  d   = (vUv - uBH.xy) * vec2(uAspect, 1.0);
  float r   = max(length(d), 1e-5), rE = uBHr;
  float win = smoothstep(rE * 16.0, rE * 4.0, r);              // fade the lens in near the hole
  uv   = uBH.xy + d / r * mix(r, r - rE * rE / r, win) / vec2(uAspect, 1.0);
  float rs = rE * 0.74;                                        // the shadow
  hole = smoothstep(rs * 0.965, rs * 1.035, r);
  ring = exp(-pow((r - rs * 1.08) / (rE * 0.045), 2.0));       // the Einstein ring
}
```

**Where is Saturn, really?** The Sky Wheel solves Kepler's equation with Newton's method, then turns JPL's orbital elements into a position.

```js
// Newton's method on Kepler's equation: E − e·sin(E) = M
const kepler = (M, e) => {
  let E = M + e * Math.sin(M);
  for (let i = 0; i < 12; i++) {
    const d = (E - e * Math.sin(E) - M) / (1 - e * Math.cos(E));
    E -= d;
    if (Math.abs(d) < 1e-12) break;
  }
  return E;
};

function helio(name, T) {                 // T = Julian centuries since J2000
  const [b, r] = EL[name];                // JPL elements and their rates per century
  const a = b[0] + r[0] * T, e = b[1] + r[1] * T, I = (b[2] + r[2] * T) * D2R;
  const L = b[3] + r[3] * T, wb = b[4] + r[4] * T, O = b[5] + r[5] * T;
  const w = (wb - O) * D2R, Om = O * D2R, M = norm(L - wb) * D2R;
  const E = kepler(M > Math.PI ? M - 2 * Math.PI : M, e);
  const xp = a * (Math.cos(E) - e), yp = a * Math.sqrt(1 - e * e) * Math.sin(E);
  // ...then rotate from the orbit's own plane into the ecliptic
}
```

**A demo mode that never pretends.** Same ticker, same numbers, every time: a hash, not a random number. The page labels it as demo data, so nobody mistakes it for a live quote.

```python
class ResearchProvider(Protocol):
    def research(self, ticker: str) -> dict[str, Any]: ...


class DemoResearchProvider:
    def _score(self, ticker: str, salt: str, low: int, high: int) -> int:
        digest = hashlib.sha256(f"{ticker}:{salt}".encode()).digest()
        return low + int.from_bytes(digest[:2], "big") % (high - low + 1)

    def research(self, ticker: str) -> dict[str, Any]:
        ticker = normalize_ticker(ticker)
        composite = self._score(ticker, "composite", 38, 92)
        posture = "Constructive" if composite >= 70 else "Mixed" if composite >= 50 else "Defensive"
        return {"ticker": ticker, "source": "demo", "source_label": "Deterministic demo data", ...}
```

def lon(body, t):
    if body == ae.Body.Sun:
        return ae.SunPosition(t).elon % 360
    if body == ae.Body.Moon:
        return ae.EclipticGeoMoon(t).lon % 360
    return ae.Ecliptic(ae.GeoVector(body, t, True)).elon % 360

speed = signed(lon(body, t.AddDays(0.5)) - lon(body, t.AddDays(-0.5)))
